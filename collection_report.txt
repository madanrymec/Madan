@app.route('/api/collection/value')
def get_collection_value():
    selected_date = request.args.get('date')
    if not selected_date:
        return jsonify({"error": "Date is required"}), 400

    conn = get_db_connection()
    cursor = conn.cursor()

    try:

        query = """
            SELECT SUM(Balance)  AS total_value 
            FROM collections_set 
            WHERE AS_ON_Date = %s AND Special_Transaction IN('COLLECTABLE DUE','COLLECTABLE NOT DUE','RETENTION NOT DUE','LC PMT DUE','LC PMT NOT DUE')
        """
        cursor.execute(query, (selected_date,))
        result = cursor.fetchone()

        })

    finally:
        cursor.close()
        conn.close()

@app.route('/api/collections/branches')
def get_branches_value():
    selected_date = request.args.get('date')
    if not selected_date:
        return jsonify({"error": "Date is required"}), 400

    conn = get_db_connection()
    cursor = conn.cursor()

    try:

        query = """
            SELECT 
                Description, 
                SUM(Balance) / 100000 AS total_value 
            FROM collections_set 
            WHERE AS_ON_Date = %s 
            AND Special_Transaction IN (
                'COLLECTABLE DUE', 
                'COLLECTABLE NOT DUE', 
                'RETENTION NOT DUE', 
                'LC PMT DUE', 
                'LC PMT NOT DUE'
            )
            GROUP BY Description
            ORDER BY total_value DESC
        """

        cursor.execute(query, (selected_date,))
        results = cursor.fetchall()
        ordered_labels = []
        ordered_values = []

        for row in results:
            ordered_labels.append(row['Description'])
            val = row['total_value']
            ordered_values.append(float(val) if val is not None else 0.0)

        return jsonify({
            "labels": ordered_labels,
            "values": ordered_values
        })

    except Exception as e:
        print(f"Error: {e}")
        return jsonify({"error": "Internal Server Error"}), 500
    finally:
        cursor.close()
        conn.close()

@app.route('/api/collections/mapped-details')
def get_mapped_collection_details():
    selected_date = request.args.get('date')
    selected_branch = request.args.get('branch')

    if not selected_date:
        return jsonify({"error": "Date is required"}), 400

    conn = get_db_connection()
    cursor = conn.cursor()

    try:
        branch_clause_set = ""
        branch_clause_data = ""
        params_set = [selected_date]
        params_data = [selected_date]

        if selected_branch and selected_branch != 'All':
            branch_clause_set = "AND Description = %s "
            branch_clause_data = "AND Coll_BR_DESC = %s "
            params_set.append(selected_branch)
            params_data.append(selected_branch)

        # 1. Fetch Outstanding - Grouped by Customer AND Profit Center
        cursor.execute(f"""
            SELECT Name1, Profit_Center, Description as branch, SUM(Balance) AS outstanding 
            FROM collections_set 
            WHERE AS_ON_Date = %s {branch_clause_set}
            AND Special_Transaction IN ('COLLECTABLE DUE', 'COLLECTABLE NOT DUE', 'RETENTION NOT DUE', 'LC PMT DUE', 'LC PMT NOT DUE', 'PMT_ADVANCE','RETENTION DUE' ) 
            GROUP BY Name1, Profit_Center, Description
        """, tuple(params_set))
        outstanding_rows = cursor.fetchall()

        # 2. Fetch Collected
        cursor.execute(f"""
            SELECT 
                Cust_Name, 
                PROFIT_CENTRE_1,
                SUM(SAP_COLL_AMT) AS collected
            FROM collections_data 
            WHERE  SAP_COLL_AMT > 0 AND Posting_Date = %s {branch_clause_data}
            GROUP BY Cust_Name, PROFIT_CENTRE_1
        """, tuple(params_data))
        collected_rows = cursor.fetchall()

        collection_map = {}
        for c in collected_rows:
            key = (c['Cust_Name'], c['PROFIT_CENTRE_1'])
            collection_map[key] = float(c['collected'] or 0)

        merged_results = []

        for row in outstanding_rows:
            name = row['Name1']
            pc = row['Profit_Center']
            out_val = float(row['outstanding'] or 0)

            key = (name, pc)
            col_amt = collection_map.get(key, 0.0)

            if col_amt > 0:
                merged_results.append({
                    "branch": row['branch'],
                    "customer": name,
                    "PROFIT_CENTRE_1": pc,
                    "outstanding": out_val,
                    "collected": col_amt,
                    "balance": out_val - col_amt
                })

        merged_results.sort(key=lambda x: x['branch'], reverse=True)

        return jsonify(merged_results)

    except Exception as e:
        print(f"Backend Error: {e}")
        return jsonify({"error": "Internal Server Error"}), 500
    finally:
        cursor.close()
        conn.close()


@app.route('/api/export/collections_excel')
def export_collections_excel():
    date_str = request.args.get('date')
    if not date_str:
        return "Please select a date", 400
    try:
        selected_date = datetime.strptime(date_str, '%Y-%m-%d').date()
    except ValueError:
        return "Invalid date format. Use YYYY-MM-DD", 400
    if selected_date == datetime.today().date():
        fetch_date = selected_date - timedelta(days=1)
    else:
        fetch_date = selected_date

    try:
        with engine.connect() as conn:

            query_coll = text("SELECT * FROM collections_data WHERE DATE(AS_ON_Date) = :dt")
            df_coll = pd.read_sql(query_coll, conn, params={"dt": fetch_date})

            query_out = text("SELECT * FROM collections_set WHERE DATE(As_on_date) = :dt")
            df_out = pd.read_sql(query_out, conn, params={"dt": fetch_date})

        if df_coll.empty and df_out.empty:
            return f"No  records found for {selected_date}", 404

        if not df_coll.empty and 'AS_ON_Date' in df_coll.columns:
            df_coll['AS_ON_Date'] = pd.to_datetime(df_coll['AS_ON_Date'], errors='coerce').dt.strftime('%d-%m-%Y')

        if not df_coll.empty and 'Document_Date' in df_coll.columns:
            df_coll['Document_Date'] = pd.to_datetime(df_coll['Document_Date'], errors='coerce').dt.strftime('%d-%m-%Y')

        if not df_coll.empty and 'Posting_Date' in df_coll.columns:
            df_coll['Posting_Date'] = pd.to_datetime(df_coll['Posting_Date'], errors='coerce').dt.strftime('%d-%m-%Y')

        if not df_coll.empty and 'Invoice_Date' in df_coll.columns:
            df_coll['Invoice_Date'] = pd.to_datetime(df_coll['Invoice_Date'], errors='coerce').dt.strftime('%d-%m-%Y')

        if not df_out.empty and 'AS_ON_Date' in df_out.columns:
            df_out['AS_ON_Date'] = pd.to_datetime(df_out['AS_ON_Date'], errors='coerce').dt.strftime('%d-%m-%Y')

        if not df_out.empty and 'Invoice_Date' in df_out.columns:
            df_out['Invoice_Date'] = pd.to_datetime(df_out['Invoice_Date'], errors='coerce').dt.strftime('%d-%m-%Y')

        if not df_out.empty and 'Baseline_Payment_Dte' in df_out.columns:
            df_out['Baseline_Payment_Dte'] = pd.to_datetime(df_out['Baseline_Payment_Dte'],
                                                            errors='coerce').dt.strftime('%d-%m-%Y')

        if not df_out.empty and 'Purchase_order_date' in df_out.columns:
            df_out['Purchase_order_date'] = pd.to_datetime(df_out['Purchase_order_date'], errors='coerce').dt.strftime(
                '%d-%m-%Y')

        if not df_out.empty and 'Posting_Date' in df_out.columns:
            df_out['Posting_Date'] = pd.to_datetime(df_out['Posting_Date'], errors='coerce').dt.strftime('%d-%m-%Y')

        output = io.BytesIO()

        with pd.ExcelWriter(output, engine='openpyxl') as writer:
            df_out.to_excel(writer, index=False, sheet_name='OUTSTANDING')
            df_coll.to_excel(writer, index=False, sheet_name='COLLECTION')

        output.seek(0)

        return send_file(
            output,
            mimetype='application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
            as_attachment=True,
            download_name=f"ZCM_Report_{selected_date}.xlsx"
        )

    except Exception as e:
        print(f"Combined Export Error: {e}")
        return f"Database Error: {str(e)}", 500

                dropdowns[j].querySelector('span').textContent = "▼";
            }

            if (!isOpen) {
                this.classList.add("active");
                dropdownContent.style.display = "block";
                this.querySelector('span').textContent = "▲";
            }
        });
    }

    // --- Auto-Open Sidebar Logic ---
    const subLinks = document.querySelectorAll('.sub-link');
    subLinks.forEach(link => {
        const linkPath = new URL(link.href).pathname;
        if (currentUrl === linkPath) {
            link.classList.add('active');
            const parentContainer = link.closest('.dropdown-container');
            const parentBtn = parentContainer.previousElementSibling;
            parentContainer.style.display = "block";
            parentBtn.classList.add('active');
            parentBtn.querySelector('span').textContent = "▲";
        }
    });
});

async function refreshDashboard() {
    showLoader();
    var selectedDate = document.getElementById('selected_date').value;
    if (!selectedDate) return;

    await updateCreditTable(selectedDate);
    try {
        //  Stock KPI
        const kpiRes = await fetch(`/api/collection/value?date=${selectedDate}`);
        const kpiData = await kpiRes.json();
        const fullStockValue = (kpiData.total_value || 0);
        document.getElementById('kpi-value').textContent = currencyFormatter.format(fullStockValue);

        const collRes = await fetch(`/api/collected/value?date=${selectedDate}`);
        const collData = await collRes.json();
        const totalCollValue = (collData.collected_value || 0);
        document.getElementById('coll-value').textContent = currencyFormatter.format(totalCollValue);


        // Product Wise Stock Value
        const stockRes = await fetch(`/api/collections/branches?date=${selectedDate}`);
        const stockData = await stockRes.json();
        renderProductStockChart(stockData);

    } catch (error) {
        console.error("Error refreshing dashboard:", error);
    }
    handleApplyFilter();
    loadBranches();
}

function getPreviousDate(inputDate) {

async function updateCreditTable(selectedDate, selectedBranch) {
    try {
        // Updated URL to include branch parameter
        const res = await fetch(`/api/collections/mapped-details?date=${selectedDate}&branch=${selectedBranch}`);
        const data = await res.json();

function filterTable() {
   var table = document.getElementById('credit-table-body');
   var rows = table.getElementsByTagName('tr');
   var filters = new Array();
   for (var i = 0; i < 14; i++) {
     var input = document.getElementById('filter' + i);
     filters[i] = input ? input.value.toLowerCase() : '';
   }
   for (var i = 0; i < rows.length; i++) {
     var row = rows[i];
     var cells = row.getElementsByTagName('td');
     var showRow = true;
     for (var j = 0; j < 14; j++) {
       if (filters[j] && cells[j]) {
         var cellText = cells[j].innerText || cells[j].textContent;
         if (cellText.toLowerCase().indexOf(filters[j]) == -1) {
           showRow = false;
         }
       }
     }
     row.style.display = showRow ? '' : 'none';
   }
 }

 function handleFilterKeypress(e) {
   if (!e) e = window.event;
   var keyCode = e.keyCode || e.which;
   if (keyCode == 13) {
     filterTable();
   }
 }

 function attachFilterEvents() {
   for (var i = 0; i < 14; i++) {
     var input = document.getElementById('filter' + i);
     if (input) {
       if (input.attachEvent) {
         input.attachEvent('onkeypress', handleFilterKeypress);
       } else {
         input.addEventListener('keypress', handleFilterKeypress);
       }
     }
   }
 }
