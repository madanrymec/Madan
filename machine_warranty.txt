# Helper function to handle email sending with the attachment
def send_email_with_pdf(to_email, customer_name, pdf_path):
    if not to_email:
        return

    if not os.path.exists(pdf_path):
        print(f"[ERROR] PDF file not found at: {pdf_path}")
        return

    try:
        msg = MIMEMultipart()
        msg['From'] = SMTP_USER
        msg['To'] = to_email
        msg['Subject'] = "Your  Product Warranty Card"

        body = f"Dear {customer_name if customer_name else 'Customer'},\n\nPlease find your attached warranty card."
        msg.attach(MIMEText(body, 'plain'))

        with open(pdf_path, 'rb') as f:
            part = MIMEApplication(f.read(), _subtype="pdf")
            part.add_header('Content-Disposition', 'attachment', filename=os.path.basename(pdf_path))
            msg.attach(part)

        server = smtplib.SMTP(SMTP_SERVER, SMTP_PORT)
        server.starttls()
        server.login(SMTP_USER, SMTP_PASS)
        server.send_message(msg)
        server.quit()
        print(f"[SUCCESS] Email sent to {to_email} with file {pdf_path}")
    except Exception as e:
        print(f"[ERROR] Email Dispatch Failed: {e}")

# API: Customer Verify
@app.route('/api/customer/verify', methods=['POST'])
def customer_verify():
    data = request.json or {}
    invoice = data.get('invoice_number', '').strip()
    serial = data.get('serial_number', '').strip()

    if not invoice or not serial:
        return jsonify({"success": False, "message": "Both Invoice and Serial Number are required."}), 400

    conn2 = get_db2()
    try:
        with conn2.cursor() as cursor2:
            cursor2.execute(
                "SELECT * FROM machine_documents WHERE invoice_number=%s AND serial_number=%s",
                (invoice, serial)
            )
            product = cursor2.fetchone()

            if not product:
                return jsonify({"success": False, "message": "Serial or Invoice Number not found in our records."}), 404

            if not product.get('warranty_card'):
                return jsonify({"success": False, "message": "The warranty card has not been uploaded yet."}), 400

        return jsonify({"success": True, "serial_number": serial})
    finally:
        conn2.close()

# API: Customer Registration / Fetch Document
@app.route('/api/customer/register', methods=['POST'])
def customer_register():
    data = request.json or {}
    serial_number = data.get('serial_number', '').strip()
    name = data.get('customer_name', '').strip() or None
    email = data.get('email', '').strip() or None
    mobile = data.get('mobile', '').strip() or None

    if not serial_number:
        return jsonify({"success": False, "message": "Serial Number is missing."}), 400

    conn2 = get_db2()
    conn1 = get_db()
    try:
        with conn2.cursor() as cursor2:
            cursor2.execute("SELECT * FROM machine_documents WHERE serial_number = %s", (serial_number,))
            product = cursor2.fetchone()

            if not product or not product.get('warranty_card'):
                return jsonify({"success": False, "message": "Invalid product record or missing warranty card filename."}), 404

            pdf_filename = product['warranty_card']

        with conn1.cursor() as cursor1:
            sql = """
                INSERT INTO customer_registrations (serial_number, customer_name, customer_email, customer_mobile, registered_at)
                VALUES (%s, %s, %s, %s, %s)
            """
            cursor1.execute(sql, (serial_number, name, email, mobile, datetime.now()))
            conn1.commit()
        pdf_path = os.path.join(LEGACY_PDF_FOLDER, pdf_filename)
        if email:
            send_email_with_pdf(email, name, pdf_path)

        return jsonify({
            "success": True,
            "message": "Document retrieved successfully!",
            "download_url": f"/download/{pdf_filename}"
        })
    except Exception as e:
        return jsonify({"success": False, "message": f"Server error: {str(e)}"}), 500
    finally:
        conn2.close()
        conn1.close()

<script>
    let verifiedSerialNumber = null;

    function toggleLoader(show) {
        document.getElementById('loader').classList.toggle('hidden', !show);
    }

    function showAlert(msg, isSuccess = false) {
        const el = document.getElementById('statusAlert');
        el.style.display = 'block';
        el.innerText = msg;
        el.style.background = isSuccess ? '#d4edda' : '#f8d7da';
        el.style.color = isSuccess ? '#155724' : '#721c24';
    }

    async function verifyProduct() {
        const invoice_number = document.getElementById('invoice_number').value.trim();
        const serial_number = document.getElementById('serial_number').value.trim();

        if (!invoice_number || !serial_number) {
            showAlert("Please enter both Invoice Number and Serial Number.", false);
            return;
        }

        toggleLoader(true);
        try {
            const res = await fetch('/api/customer/verify', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ invoice_number, serial_number })
            });
            const data = await res.json();

            if (data.success) {
                verifiedSerialNumber = data.serial_number;
                document.getElementById('step1').classList.add('hidden');
                document.getElementById('step2').classList.remove('hidden');
                showAlert('Product verified! Enter details to proceed.', true);
            } else {
                showAlert(data.message, false);
            }
        } catch (err) {
            showAlert("Server error during verification. Please try again.", false);
        } finally {
            toggleLoader(false);
        }
    }

    async function registerWarranty() {
        const customer_name = document.getElementById('customer_name').value.trim();
        const email = document.getElementById('email').value.trim();
        const mobile = document.getElementById('mobile').value.trim();

        toggleLoader(true);
        try {
            const res = await fetch('/api/customer/register', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({
                    serial_number: verifiedSerialNumber,
                    customer_name,
                    email,
                    mobile
                })
            });
            const data = await res.json();

            if (data.success) {
                document.getElementById('step2').classList.add('hidden');
                document.getElementById('step3').classList.remove('hidden');
                document.getElementById('statusAlert').style.display = 'none';
            } else {
                showAlert(data.message, false);
            }
        } catch (err) {
            showAlert("Server error during processing. Please try again.", false);
        } finally {
            toggleLoader(false);
        }
    }
</script>
