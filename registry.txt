function downloadPurchaseExcel() {
    const params = new URLSearchParams();

    const fyVal = document.getElementById('f-fin-year').value;
    const vnVal = document.getElementById('f-vendor').value;
    const dtVal = document.getElementById('f-date').value;

    // 1. Prioritize Exact Date / Date Range if it's filled out
    if (dtVal) {
        params.append('date', dtVal);
    }
    // 2. Otherwise, fallback to the default Year & Month filters
    else {
        if (fyVal) params.append('fin_year', fyVal);

        if (state.months && state.months.length > 0) {
            state.months.forEach(m => params.append('month', m));
        } else if (!fyVal) {
            alert("Please select either a Year/Month or a Date Range to export data.");
            return;
        }
    }

    // Append standard categorical filters
    if (vnVal) params.append('vendor', vnVal);

    // Note: Plant uses the pill design, so we pull from the state variable
    if (state.plant) params.append('plant', state.plant);

    // Trigger the download
    const qs = params.toString();
    window.location.href = `/api/export/purchase_excel?${qs}`;
}

    // --- Cascading Vendor Dropdown Updater ---
    async function updateVendorFilter() {
        const params = new URLSearchParams();
        const fy = document.getElementById('f-fin-year').value;
        const dt = document.getElementById('f-date').value;

        if (fy) params.append('fin_year', fy);
        if (dt) params.append('date', dt);
        if (state.plant) params.append('plant', state.plant);
        state.months.forEach(m => params.append('month', m));

        try {
            const res = await fetch(`/api/filters/purchase?${params.toString()}`);
            if (!res.ok) return;
            const data = await res.json();

            const venDrop = document.getElementById('f-vendor');
            const selectedVendor = venDrop.value;

            venDrop.innerHTML = '<option value="">All Vendors</option>';
            if (data.vendors) {
                data.vendors.forEach(v => venDrop.add(new Option(v, v)));
            }

            // Retain currently selected vendor if it exists in the updated list
            if ([...venDrop.options].some(opt => opt.value === selectedVendor)) {
                venDrop.value = selectedVendor;
            } else {
                venDrop.value = '';
            }
        } catch (e) {
            console.error("Vendor filter update error:", e);
        }
    }

        const monthChecks = document.querySelectorAll('.month-check');

            monthSelectBox.textContent = 'All Months';

            monthSelectBox.textContent = checked.join(', ');
        } else {
            monthSelectBox.textContent = `${checked.length} Selected`;
        }

        if (triggerVendorUpdate) {

def build_purchase_filter(exclude_vendor=False):
    """Helper function to build SQL WHERE clause for Purchase queries, supporting date ranges and cascading vendor filter."""
    filters = []
    params = []

    fin_years = request.args.getlist('fin_year')
    months = request.args.getlist('month')
    plants = request.args.getlist('plant')
    vendors = request.args.getlist('vendor')
    exact_date = request.args.get('date')

    if fin_years:
        placeholders = ','.join(['%s'] * len(fin_years))
        filters.append(f"fin_Year IN ({placeholders})")
        params.extend(fin_years)

    if months:
        month_map = {
            "January": 1, "February": 2, "March": 3, "April": 4, "May": 5, "June": 6,
            "July": 7, "August": 8, "September": 9, "October": 10, "November": 11, "December": 12
        }
        month_nums = [str(month_map[m]) for m in months if m in month_map]
        if month_nums:
            placeholders = ','.join(['%s'] * len(month_nums))
            filters.append(f"MONTH(Post_Date) IN ({placeholders})")
            params.extend(month_nums)

    if plants:
        placeholders = ','.join(['%s'] * len(plants))
        filters.append(f"ValA IN ({placeholders})")
        params.extend(plants)

    # Omit vendor constraint when updating the vendor dropdown itself
    if not exclude_vendor and vendors:
        placeholders = ','.join(['%s'] * len(vendors))
        filters.append(f"Vendor_Name IN ({placeholders})")
        params.extend(vendors)

    # Date Range Logic
    if exact_date:
        if " to " in exact_date:
            start_date, end_date = exact_date.split(" to ")
            filters.append("DATE(Post_Date) BETWEEN %s AND %s")
            params.extend([start_date, end_date])
        else:
            filters.append("DATE(Post_Date) = %s")
            params.append(exact_date)

    where_clause = "WHERE " + " AND ".join(filters) if filters else ""
    return where_clause, params


@app.route('/api/filters/purchase', methods=['GET'])
def get_purchase_filters():
    """Fetches unique values for the Purchase Dashboard filters with cascading vendor lookup."""
    # Build filter using date parameters and plant, excluding vendor
    vendor_where, vendor_params = build_purchase_filter(exclude_vendor=True)

    with get_db_connection() as conn:
        with conn.cursor() as cursor:
            # Independent: Financial Years & Plants
            cursor.execute("SELECT DISTINCT fin_Year FROM purchase_registry WHERE fin_Year IS NOT NULL ORDER BY fin_Year DESC")
            fin_years = [r['fin_Year'] for r in cursor.fetchall() if r['fin_Year']]

            cursor.execute("SELECT DISTINCT ValA FROM purchase_registry WHERE ValA IS NOT NULL ORDER BY ValA ASC")
            plants = [r['ValA'] for r in cursor.fetchall() if r['ValA']]

            # Dependent: Vendors matching selected Year, Month, Date Range, and Plant
            if vendor_where:
                vendor_query = f"SELECT DISTINCT Vendor_Name FROM purchase_registry {vendor_where} AND Vendor_Name IS NOT NULL ORDER BY Vendor_Name ASC"
            else:
                vendor_query = "SELECT DISTINCT Vendor_Name FROM purchase_registry WHERE Vendor_Name IS NOT NULL ORDER BY Vendor_Name ASC"

            cursor.execute(vendor_query, vendor_params)
            vendors = [r['Vendor_Name'] for r in cursor.fetchall() if r['Vendor_Name']]

    return jsonify({"fin_years": fin_years, "plants": plants, "vendors": vendors})
